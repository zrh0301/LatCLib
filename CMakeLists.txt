cmake_minimum_required(VERSION 3.16)

project(MyPQC VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(PQC_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib")
set(PQC_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PQC_LIBRARY_OUTPUT_DIRECTORY}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PQC_LIBRARY_OUTPUT_DIRECTORY}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PQC_RUNTIME_OUTPUT_DIRECTORY}")

message("\n-- Configuring ${PROJECT_NAME} ${PROJECT_VERSION}...\n")

message(STATUS "Available architectures (default = AUTO):\n")
message("   PQC_ARCH=AUTO   Detect the architecture from CMAKE_SYSTEM_PROCESSOR.")
message("   PQC_ARCH=X64    AMD x86_64-compatible 64-bit architecture.")
message("   PQC_ARCH=A64    ARM 64-bit architecture.\n")

message(STATUS "Available optimized backends (default = AUTO):\n")
message("   PQC_BACKEND=AUTO       Backend matching the target architecture.")
message("   PQC_BACKEND=AVX2       x86-64 AVX2 implementation.")
message("   PQC_BACKEND=GFNI       x86-64 AVX2 + GFNI implementation.")
message("   PQC_BACKEND=NEON       AArch64 NEON implementation.\n")

message(STATUS "Available modules (default = ALL):\n")
message("   PQC_WITH=DILITHIUM   Dilithium signatures and K-PKE helpers.")
message("   PQC_WITH=KYBER       Kyber K-PKE.")
message("   PQC_WITH=SNOVA_R2    SNOVA round 2 signatures.")
message("   PQC_WITH=SNOVA_R3    SNOVA round 3 signatures.")
message("   PQC_WITH=MAYO        MAYO signatures.")
message("   PQC_WITH=MQOM        MQOM signatures.")
message("   PQC_WITH=NGCC_DOVE   NGCC round-1 DOVE signatures.")
message("   PQC_WITH=NGCC_VDOO   NGCC round-1 VDOO signatures.")
message("   PQC_WITH=NGCC_FLEXTREE NGCC round-1 FlexTree signatures.")
message("   PQC_WITH=NGCC_AIGIS  NGCC round-1 Aigis-Enc+ KEM.")
message("   PQC_WITH=ALL         All of the above.\n")

message(STATUS "Available switches (default = SHLIB, STLIB, TESTS, INSTALL):\n")
message("   PQC_SHLIB=[off|on]     Build shared libraries.")
message("   PQC_STLIB=[off|on]     Build static libraries.")
message("   PQC_TESTS=[off|on]     Build automated tests.")
message("   PQC_PERF_TESTS=[off|on] Enable performance tests in CTest.")
message("   PQC_INSTALL=[off|on]   Generate installation rules.")
message("   PQC_NATIVE=[off|on]    Optimize for the build host CPU.")
message("   PQC_SANITIZERS=[off|on] Build with ASan and UBSan.")
message("   PQC_STRIP=[off|on]     Remove unused code and data.")
message("   PQC_AES=AUTO|C|NI      AES backend.\n")

message(STATUS "Number of repetitions and extra flags:\n")
message("   PQC_CFLAGS=<flags>     Extra compiler flags appended to every target.\n")

include(cmake/options.cmake)
include(cmake/arch.cmake)
include(cmake/flags.cmake)
include(cmake/functions.cmake)
include(cmake/gf.cmake)
include(cmake/algorithms.cmake)

configure_file(inc/core/pqc_conf.h.in
    "${PROJECT_BINARY_DIR}/include/core/pqc_conf.h" @ONLY)

if(PQC_TESTS)
    enable_testing()
    include(cmake/tests.cmake)
    add_subdirectory(Test)
endif()

if(PQC_INSTALL)
    include(cmake/install.cmake)
endif()

message(STATUS "")
message(STATUS "Configured architecture: ${PQC_ARCH_EFFECTIVE}")
message(STATUS "Configured backend: ${PQC_BACKEND_EFFECTIVE}")
message(STATUS "Configured GF backend: ${PQC_GF_BACKEND_EFFECTIVE}")
message(STATUS "Configured AES backend: ${PQC_AES_EFFECTIVE}")
message(STATUS "Configured modules: ${PQC_WITH_EFFECTIVE}")
set(PQC_LIBRARY_KINDS "")
if(PQC_SHLIB)
    string(APPEND PQC_LIBRARY_KINDS "shared ")
endif()
if(PQC_STLIB)
    string(APPEND PQC_LIBRARY_KINDS "static")
endif()
message(STATUS "Libraries: ${PQC_LIBRARY_KINDS}")
message(STATUS "")
